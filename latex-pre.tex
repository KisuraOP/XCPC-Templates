% 底色向下多画 2 pt，填满字体撑高的行间空隙；盒子尺寸不变。
\makeatletter
\let\lst@bgblock\lst@frameBlock
\def\lst@frameBlock#1#2#3#4{%
    \ifx#1\lst@bkgcolor
        \begingroup
        \setbox\@tempboxa\hbox{%
            \lst@bgblock{#1}{#2}{#3}{\dimexpr#4+2pt\relax}%
        }%
        \dp\@tempboxa=#4\relax
        \box\@tempboxa
        \endgroup
    \else
        \lst@bgblock{#1}{#2}{#3}{#4}%
    \fi
}
\makeatother
